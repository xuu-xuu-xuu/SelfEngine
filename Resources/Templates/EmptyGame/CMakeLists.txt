cmake_minimum_required(VERSION 3.25)
project(EmptyGame LANGUAGES C CXX)
set(SELF_ENGINE_ROOT "" CACHE PATH "SelfEngine source directory")
if(NOT EXISTS "${SELF_ENGINE_ROOT}/Source/ToolLayer/Tools/Project.h")
    message(FATAL_ERROR "Set SELF_ENGINE_ROOT to the SelfEngine source directory")
endif()
set(BUILD_TESTING OFF CACHE BOOL "Build engine tests" FORCE)
add_subdirectory("${SELF_ENGINE_ROOT}" "${CMAKE_BINARY_DIR}/engine")
add_executable(EmptyGameHeadless Main.cpp)
target_link_libraries(EmptyGameHeadless PRIVATE EngineProject)
target_compile_features(EmptyGameHeadless PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(EmptyGameHeadless PRIVATE /utf-8 /W4)
endif()
